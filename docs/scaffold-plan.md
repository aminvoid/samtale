## Step 8: Environment schema, headers, utilities

`src/lib/env/schema.ts` (pure; imported by `next.config.ts` and scripts, so **no** `server-only` here):

```ts
import { z } from "zod";

export const buildEnvSchema = z
  .object({
    CONTENT_RELEASE: z.enum(["preview", "production"]).default("preview"),
    SITE_INDEXING: z.enum(["on", "off"]).default("off"),
    NEXT_PUBLIC_SITE_URL: z.url().optional(),
    AUDIO_BASE_URL: z.string().min(1).default("/audio"),
    VERCEL_ENV: z.enum(["production", "preview", "development"]).optional(),
  })
  .superRefine((env, ctx) => {
    if (env.VERCEL_ENV !== "production") return;
    if (env.CONTENT_RELEASE !== "production") {
      ctx.addIssue({
        code: "custom",
        path: ["CONTENT_RELEASE"],
        message: "Production deployments must build with CONTENT_RELEASE=production (ADR-0010).",
      });
    }
    if (env.NEXT_PUBLIC_SITE_URL === undefined) {
      ctx.addIssue({ code: "custom", path: ["NEXT_PUBLIC_SITE_URL"], message: "Required in production (ADR-0010)." });
    }
  });

export type BuildEnv = z.infer<typeof buildEnvSchema>;

export function parseBuildEnv(source: Record<string, string | undefined>): BuildEnv {
  const result = buildEnvSchema.safeParse(source);
  if (!result.success) {
    throw new Error(`Invalid environment variables:\n${z.prettifyError(result.error)}`);
  }
  return result.data;
}
```

`src/lib/security/headers.ts` (ADR-0011):

```ts
export type HeaderPair = { key: string; value: string };

export function buildContentSecurityPolicy(options: { isDev: boolean; onHttpsHost: boolean }): string {
  const { isDev, onHttpsHost } = options;
  const directives: Array<[string, string[]]> = [
    ["default-src", ["'self'"]],
    ["script-src", ["'self'", "'unsafe-inline'", ...(isDev ? ["'unsafe-eval'"] : [])]],
    ["style-src", ["'self'", "'unsafe-inline'"]],
    ["img-src", ["'self'", "data:", "blob:"]],
    ["font-src", ["'self'"]],
    ["media-src", ["'self'", "blob:"]],
    ["connect-src", ["'self'", ...(isDev ? ["ws:"] : [])]],
    ["worker-src", ["'self'"]],
    ["manifest-src", ["'self'"]],
    ["object-src", ["'none'"]],
    ["base-uri", ["'self'"]],
    ["form-action", ["'self'"]],
    ["frame-ancestors", ["'none'"]],
  ];
  const parts = directives.map(([name, values]) => `${name} ${values.join(" ")}`);
  if (onHttpsHost) parts.push("upgrade-insecure-requests");
  return parts.join("; ");
}

export const PERMISSIONS_POLICY = [
  "microphone=(self)",
  "camera=()",
  "geolocation=()",
  "display-capture=()",
  "usb=()",
  "serial=()",
  "hid=()",
  "bluetooth=()",
  "midi=()",
  "payment=()",
  "magnetometer=()",
  "gyroscope=()",
  "accelerometer=()",
  "browsing-topics=()",
  "interest-cohort=()",
].join(", ");

export function buildSecurityHeaders(options: {
  isDev: boolean;
  indexing: "on" | "off";
  onHttpsHost: boolean;
}): HeaderPair[] {
  const headers: HeaderPair[] = [
    { key: "Content-Security-Policy", value: buildContentSecurityPolicy(options) },
    { key: "Permissions-Policy", value: PERMISSIONS_POLICY },
    { key: "Referrer-Policy", value: "same-origin" },
    { key: "X-Content-Type-Options", value: "nosniff" },
    { key: "X-Frame-Options", value: "DENY" },
    { key: "Cross-Origin-Opener-Policy", value: "same-origin" },
    { key: "Cross-Origin-Resource-Policy", value: "same-origin" },
  ];
  if (options.onHttpsHost) {
    headers.push({ key: "Strict-Transport-Security", value: "max-age=63072000; includeSubDomains" });
  }
  if (options.indexing === "off") {
    headers.push({ key: "X-Robots-Tag", value: "noindex, nofollow" });
  }
  return headers;
}

export const AUDIO_CACHE_HEADERS: HeaderPair[] = [
  { key: "Cache-Control", value: "public, max-age=31536000, immutable" },
];
```

import type { AppLocale } from "./routing";

const RTL_LANGUAGES = new Set(["ar", "ckb", "dv", "fa", "he", "ps", "sd", "ug", "ur", "yi"]);

export type Direction = "ltr" | "rtl";

/** The only place direction is decided (ADR-0004 §2). */
export function getDirection(locale: string): Direction {
  return RTL_LANGUAGES.has(new Intl.Locale(locale).language) ? "rtl" : "ltr";
}

/**
 * Digit style (D-2). Empty = CLDR default per locale: fa → "arabext" (۰–۹), en → "latn".
 * To show Latin digits in the Persian interface, change to: { fa: "latn" }.
 */
export const NUMBERING_SYSTEM_OVERRIDES: Partial<Record<AppLocale, string>> = {};

export function numberingSystemFor(locale: AppLocale): string {
  return NUMBERING_SYSTEM_OVERRIDES[locale] ?? new Intl.NumberFormat(locale).resolvedOptions().numberingSystem;
}

/** Proposed: Gregorian dates in Persian words and digits. Owner to confirm (ADR-0004 §5). */
export const CALENDAR = "gregory";

export const TIME_ZONE = "Europe/Copenhagen";
```

import type { Formats } from "next-intl";
import { CALENDAR, numberingSystemFor } from "./config";
import type { AppLocale } from "./routing";

export function buildFormats(locale: AppLocale) {
  const numberingSystem = numberingSystemFor(locale);
  return {
    number: {
      integer: { maximumFractionDigits: 0, numberingSystem },
      year: { useGrouping: false, numberingSystem },
      twoDigit: { minimumIntegerDigits: 2, numberingSystem },
    },
    dateTime: {
      date: { day: "numeric", month: "long", year: "numeric", calendar: CALENDAR, numberingSystem },
      dayMonth: { day: "numeric", month: "long", calendar: CALENDAR, numberingSystem },
    },
  } satisfies Formats;
}

export type AppFormats = ReturnType<typeof buildFormats>;
```

export default getRequestConfig(async ({ locale }) => {

`scripts/ci/check-build-leaks.ts` (ADR-0002 §4.5):

```ts
/**
 * Fails if internal content appears anywhere in the Next.js build output.
 * Usage: tsx scripts/ci/check-build-leaks.ts [--production]
 */
import { existsSync, readdirSync, readFileSync, statSync } from "node:fs";
import { join, resolve } from "node:path";
import { z } from "zod";

const root = resolve(import.meta.dirname, "../..");
const production = process.argv.includes("--production");
const TEXT_FILE = /\.(js|mjs|cjs|json|html|rsc|txt|body|meta|map)$/;
const topicReviewSchema = z.object({ review: z.object({ status: z.string(), notes: z.array(z.unknown()).optional() }) });

function collectStrings(value: unknown, out: string[]): void {
  if (typeof value === "string") out.push(value);
  else if (Array.isArray(value)) value.forEach((v) => collectStrings(v, out));
  else if (value !== null && typeof value === "object") Object.values(value).forEach((v) => collectStrings(v, out));
}

function readJson(path: string): unknown {
  return existsSync(path) ? JSON.parse(readFileSync(path, "utf8")) : null;
}

// Canaries: every string from internal files and every review note in topic files.
const canaries = new Set<string>(["siriParts", "review-ledger", "needs-review"]);
const internalStrings: string[] = [];
collectStrings(readJson(join(root, "content/internal/margin-notes.json")), internalStrings);
collectStrings(readJson(join(root, "content/review-ledger.json")), internalStrings);
const topicsDir = join(root, "content/topics");
const unapprovedIds: string[] = [];
if (existsSync(topicsDir)) {
  for (const file of readdirSync(topicsDir).filter((f) => f.endsWith(".json"))) {
    const parsed = topicReviewSchema.safeParse(readJson(join(topicsDir, file)));
    if (!parsed.success) continue;
    collectStrings(parsed.data.review.notes, internalStrings);
    if (parsed.data.review.status !== "approved") unapprovedIds.push(file.replace(/\.json$/, ""));
  }
}
for (const s of internalStrings) if (s.trim().length >= 16) canaries.add(s.trim());
if (production) for (const id of unapprovedIds) canaries.add(`"${id}"`);

function* walk(dir: string): Generator<string> {
  for (const entry of readdirSync(dir)) {
    if (entry === "cache") continue;
    const path = join(dir, entry);
    if (statSync(path).isDirectory()) yield* walk(path);
    else if (TEXT_FILE.test(entry)) yield path;
  }
}

const buildDir = join(root, ".next");
if (!existsSync(buildDir)) {
  process.stderr.write("No .next build output. Run `npm run build` first.\n");
  process.exit(1);
}
const hits: string[] = [];
for (const file of walk(buildDir)) {
  const text = readFileSync(file, "utf8");
  for (const canary of canaries) if (text.includes(canary)) hits.push(`${file}: ${canary.slice(0, 60)}`);
}
if (hits.length > 0) {
  process.stderr.write(`INTERNAL CONTENT FOUND IN BUILD OUTPUT (${hits.length}):\n${hits.join("\n")}\n`);
  process.exit(1);
}
process.stdout.write(`check:leaks OK (${canaries.size} canaries${production ? ", production" : ""})\n`);
```

const files = readdirSync(DIR).filter((f) => /^[a-z]{2,3}(-[A-Z]{2})?\.json$/.test(f));

function flatten(value: unknown, prefix = "", out = new Map<string, string>()): Map<string, string> {
  if (typeof value === "string") out.set(prefix, value);
  else if (value !== null && typeof value === "object") {
    for (const [key, child] of Object.entries(value)) flatten(child, prefix === "" ? key : `${prefix}.${key}`, out);
  }
  return out;
}
const load = (file: string) => flatten(JSON.parse(readFileSync(resolve(DIR, file), "utf8")));
const argsOf = (message: string) =>
  [...message.matchAll(/\{\s*([A-Za-z_]\w*)\s*[,}]/g)].map((m) => m[1]).sort().join(",");

const fa = load("fa.json");
const PLAIN_TEXT_KEYS = /^metadata\./;

function hasLatinOutsideTags(message: string): boolean {
  const stripped = message
    .replace(/<(da|ltr)>[\s\S]*?<\/\1>/g, "")
    .replace(/\{\s*[A-Za-z_]\w*\s*(,\s*[A-Za-z]+\s*)?(,\s*[A-Za-z]+\s*)?/g, "{")
    .replace(/\b(zero|one|two|few|many|other|=\d+)\s*\{/g, "{");
  return /[A-Za-z]/.test(stripped);
}

describe("messages", () => {
  it.each(files)("%s has exactly the keys and ICU arguments of fa.json", (file) => {
    const other = load(file);
    expect([...other.keys()].sort()).toEqual([...fa.keys()].sort());
    for (const [key, message] of fa) expect(argsOf(other.get(key) ?? ""), key).toBe(argsOf(message));
  });

  it("fa: numbers use named formats, never a bare # (ADR-0004 §5)", () => {
    for (const [key, message] of fa) expect(message.includes("#"), key).toBe(false);
  });

  it("fa: Latin text appears only inside <da> or <ltr> tags (ADR-0004 §3)", () => {
    for (const [key, message] of fa) {
      if (!PLAIN_TEXT_KEYS.test(key)) expect(hasLatinOutsideTags(message), key).toBe(false);
    }
  });

  it("fa: Persian letters only (no Arabic ي/ك, no presentation forms)", () => {
    for (const [key, message] of fa) expect(/[يكﭐ-﷿ﹰ-﻿]/.test(message), key).toBe(false);
  });
});
